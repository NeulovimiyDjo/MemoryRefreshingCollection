##Setup Fedora docker storage
sudo mkdir /virt/my_user
sudo chown my_user:my_user /virt/my_user
chmod 700 /virt/my_user
getfacl /home/my_user && getfacl /virt/my_user
sudo semanage fcontext -a -e /home/my_user /virt/my_user

podman info
mkdir ~/.config/containers
nano /home/my_user/.config/containers/storage.conf
#[storage]
#driver = "overlay"
#graphroot = "/virt/my_user/.local/share/containers/storage"
sudo rm -rf /home/my_user/.local/share/containers/storage
sudo systemctl restart podman
podman info
sudo restorecon -R -v /virt/my_user
ls -laZ /virt/my_user/.local/share/containers/storage

##Setup Fedora virtualization
dnf group list --hidden
dnf group info virtualization
sudo dnf install @virtualization
sudo systemctl status libvirtd
sudo systemctl enable libvirtd
sudo systemctl start libvirtd
sudo getent group | grep libvirt
sudo usermod -a -G libvirt $(whoami)
newgrp libvirt
sudo dnf install guestfs-tools

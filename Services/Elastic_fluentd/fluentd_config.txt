
<source>
  @type tail
  path /logs/web/log*.txt
  pos_file /logcollectors_work/fluentd_posfile_web.pos
  read_from_head false
  follow_inodes true
  path_key logfile
  tag mylogs.web
  from_encoding utf-8
  encoding utf-8
  multiline_flush_interval 5s
  <parse>
    @type multiline
    format_firstline /^(?:FATAL|ERROR|WARN|INFO|DEBUG|TRACE)\|\d{4}-\d{2}-\d{2}\s\d{2}:\d{2}:\d{2}\.\d{4}\|(?:[\d\w.]*(?::[\d\w]*)?):/
    format1 /^(?<level>FATAL|ERROR|WARN|INFO|DEBUG|TRACE)\|(?<time>\d{4}-\d{2}-\d{2}\s\d{2}:\d{2}:\d{2}\.\d{3})\d{1}\|(?<logger>[\d\w.]*(?::[\d\w]*)?):\n(?<message>.*)/
    time_key time
    time_format %Y-%m-%d %H:%M:%S.%L
    keep_time_key false
  </parse>
</source>

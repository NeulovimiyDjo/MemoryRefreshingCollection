cat << EOF > 1.lua
function add_time_field(tag, timestamp, record)
    local now = os.time()
    local tz_diff_sec = os.difftime(now, os.time(os.date("!*t", now)))
    local datetime_without_ms_str = os.date("%Y-%m-%dT%H:%M:%S", timestamp - tz_diff_sec)
    local ms_str = string.match(tostring(timestamp), "%d%.(%d+)")
    record["time"] = string.format("%s.%sZ", datetime_without_ms_str, ms_str)
    return 2, timestamp, record
end
EOF
